   427→function Plane({
   428→  name,
   429→  items,
   430→  icon,
   431→  tone,
   432→  note,
   433→}: {
   434→  name: string;
   435→  items: string[];
   436→  icon: React.ReactNode;
   437→  tone: "light" | "dark";
   438→  note?: string;
   439→}) {
   440→  const dark = tone === "dark";
   441→  return (
   442→    <div
   443→      className={`rounded-2xl border p-5 ${
   444→        dark ? "border-console-line bg-console text-console-ink" : "border-border bg-card"
   445→      }`}

   447→      <div className="flex flex-wrap items-center justify-between gap-2">

   449→          <span className={dark ? "text-verdict-verified" : "text-primary"}>{icon}</span>
   450→          {name}
   451→        </p>
   452→        {note && (
   453→          <p className={`font-mono text-[10px] uppercase tracking-wider ${dark ? "text-console-ink/45" : "text-muted-foreground"}`}>{note}</p>
   454→        )}
   455→      </div>
   456→      <div className="mt-3 flex flex-wrap gap-2">
   457→        {items.map((i) => (
   458→          <span
   459→            key={i}
   460→            className={`rounded-lg border px-2.5 py-1 font-mono text-[11.5px] ${
   461→              dark ? "border-console-line bg-white/[0.04] text-console-ink/85" : "border-border bg-secondary/70 text-foreground/80"
   462→            }`}
